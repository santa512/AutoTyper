cmake_minimum_required(VERSION 3.21)
project(AutoTyperCompact VERSION 0.1.0 LANGUAGES CXX RC)

if(NOT WIN32)
    message(FATAL_ERROR "Auto Typer Compact requires a Windows toolchain.")
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

add_library(typing_engine src/TypingEngine.cpp)
target_include_directories(typing_engine PUBLIC src)
add_executable(AutoTyperCompact WIN32 src/main.cpp resources/app.rc)
target_include_directories(AutoTyperCompact PRIVATE resources)
target_compile_definitions(AutoTyperCompact PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00 WINVER=0x0A00)
target_link_libraries(AutoTyperCompact PRIVATE typing_engine user32 gdi32 comctl32 dwmapi uxtheme advapi32)
if(MSVC)
    # app.rc already embeds the DPI/UAC/common-controls manifest as resource 1.
    target_link_options(AutoTyperCompact PRIVATE /MANIFEST:NO)
    target_compile_options(typing_engine PRIVATE /W4 /utf-8)
    target_compile_options(AutoTyperCompact PRIVATE /W4 /utf-8 /permissive-)
else()
    target_compile_options(typing_engine PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(AutoTyperCompact PRIVATE -Wall -Wextra -Wpedantic)
    target_link_options(AutoTyperCompact PRIVATE -municode -static)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(engine_tests tests/engine_tests.cpp)
    target_link_libraries(engine_tests PRIVATE typing_engine)
    if(MINGW)
        target_link_options(engine_tests PRIVATE -static)
    endif()
    add_test(NAME typing_engine COMMAND engine_tests)
    add_executable(input_target WIN32 tests/input_target.cpp)
    target_compile_definitions(input_target PRIVATE UNICODE _UNICODE)
    target_link_libraries(input_target PRIVATE user32 gdi32)
    if(MINGW)
        target_link_options(input_target PRIVATE -municode -static)
    endif()
endif()

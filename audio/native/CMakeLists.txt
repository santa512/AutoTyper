cmake_minimum_required(VERSION 3.21)
project(AutoTyperAudio LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(AutoTyperAudio SHARED processor.cpp)
if(MSVC)
  target_compile_options(AutoTyperAudio PRIVATE /O2 /bigobj /EHsc)
else()
  target_compile_options(AutoTyperAudio PRIVATE -O3)
  target_link_options(AutoTyperAudio PRIVATE -static -static-libgcc -static-libstdc++)
endif()
